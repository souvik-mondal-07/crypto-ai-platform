# Step 1 + Step 2 + Step 3 + Step 4 + Phase 12 (news & sentiment).
# Future dependencies (redis, celery, xgboost, lightgbm, torch,
# transformers, news SDKs, etc.) will be added in
# later steps when the corresponding features are built. Technical
# analysis (Phase 10) deliberately does NOT add pandas/numpy/ta here —
# RSI/MACD/SMA/EMA/Bollinger/ATR are implemented as small, dependency-
# free pure-Python functions in app/services/indicators.py. Those
# formulas are standard and well-documented, so there's no accuracy
# trade-off, and it avoids pulling in a numeric-library dependency for
# what is a handful of straightforward rolling calculations.

fastapi==0.115.0
uvicorn[standard]==0.30.6
pydantic==2.9.2
pydantic-settings==2.5.2
python-dotenv==1.0.1
# email-validator backs Pydantic's EmailStr, used by auth request
# schemas (app/schemas/auth.py) for real email-format validation.
email-validator==2.2.0

# Database (Step 2) — PyMongo's native async API (AsyncMongoClient)
# requires PyMongo 4.9+; no separate ODM (e.g. Beanie/MongoEngine) or
# Motor is introduced, per the architecture decision in
# docs/database-schema.md.
pymongo==4.9.2

# Market data providers (Step 3) — async HTTP client used by
# app/providers/*/client.py to call CoinGecko's and Binance's public
# REST APIs. Never `requests` inside async routes/services.
# 0.28.x (not 0.27) because the Gemini SDK (`google-genai`, Phase 15) requires httpx>=0.28. The app only
# uses plain `httpx.AsyncClient()`, which is unchanged in 0.28; respx/pytest-httpx below are the matching versions.
httpx==0.28.1

# Authentication (Step 4)
# argon2-cffi: password hashing (app/core/security.py). No bcrypt/
# scrypt — Argon2id is the current OWASP-recommended default.
argon2-cffi==23.1.0
# python-jose: JWT creation/verification. [cryptography] extra pulls
# in `cryptography` for the actual crypto backend (avoids the
# pure-Python fallback).
python-jose[cryptography]==3.3.0

# Sentiment analysis (Phase 12) — FinBERT (ProsusAI/finbert) via Hugging Face.
# Loaded lazily and only once per process by app/services/sentiment_model.py;
# the API and news ingestion still run without these installed (articles
# then simply stay un-analyzed and sentiment reports "unavailable"). The
# model weights (~440 MB) are downloaded on first use and cached by
# Hugging Face. Use the CPU build of torch if you have no GPU:
#   pip install torch --index-url https://download.pytorch.org/whl/cpu
transformers>=4.44,<5
torch>=2.2

# ML prediction engine (Phase 13) — used by the repo-level `ml/` package, which
# the backend imports lazily for inference (see app/services/prediction_service.py).
# Training is an explicit CLI step and is never started by the API.
numpy>=1.26
pandas>=2.1
scikit-learn>=1.4
xgboost>=2.0
lightgbm>=4.3
# (torch, already listed above, backs the optional LSTM/GRU model.)

# Gemini AI analysis (Phase 15) — the official Google Gen AI SDK (the `google-genai` package,
# imported as `from google import genai`). It is imported lazily by app/services/gemini_service.py,
# so the rest of the backend runs even when it is not installed (AI analysis then reports
# "unavailable"). Not the legacy `google-generativeai` package.
google-genai>=1.0

# Testing
pytest==8.3.3
pytest-asyncio==0.24.0
pytest-httpx==0.35.0
# respx < 0.22 cannot intercept httpx 0.28 requests ("RESPX: ... not mocked!"); 0.22.0 adds httpx 0.28 support.
respx==0.22.0
